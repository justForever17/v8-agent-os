cmake_minimum_required(VERSION 3.16)
project(v8_executor_core_tests C)
enable_testing()
add_executable(executor_core_test test_core.c ../main/executor_core.c)
target_include_directories(executor_core_test PRIVATE ../main)
target_compile_options(executor_core_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(executor_core_test PRIVATE -fsanitize=address,undefined)
add_test(NAME executor_core COMMAND executor_core_test)
find_package(OpenSSL REQUIRED)
find_path(CJSON_INCLUDE_DIR cJSON.h PATH_SUFFIXES cjson REQUIRED)
find_library(CJSON_LIBRARY cjson REQUIRED)
add_executable(executor_wire_test test_wire.c ../main/wire_codec.c)
target_include_directories(executor_wire_test PRIVATE ../main ${CJSON_INCLUDE_DIR})
target_compile_definitions(executor_wire_test PRIVATE VX_HOST=1)
target_compile_options(executor_wire_test PRIVATE -Wall -Wextra -Werror -fsanitize=address,undefined)
target_link_options(executor_wire_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(executor_wire_test PRIVATE ${CJSON_LIBRARY} OpenSSL::Crypto m)
add_test(NAME executor_wire COMMAND executor_wire_test ${CMAKE_CURRENT_SOURCE_DIR}/engine-command.json)
